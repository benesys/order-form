// 주문 전송 함수 async function submitOrder() { // 위 1번에서 새로 발급받은 Apps Script 웹 앱 배포 URL을 넣어주세요. const scriptUrl = "https://script.google.com/macros/s/AKfycbzP4KiO8_TlWfx9TUOW2nuFjiAjkaPqPLL7irl45R2Y0L2bIbvGrVsYu2ZkMYXfClsr/exec"; // index.html의 입력창 id에 맞게 수집할 데이터 객체 const orderData = { name: document.getElementById("name").value, phone: document.getElementById("phone").value, product: document.getElementById("product").value, qty: document.getElementById("qty").value, address: document.getElementById("address").value }; if (!orderData.name || !orderData.phone || !orderData.address) { alert("이름, 연락처, 배송주소를 모두 입력해 주세요."); return; } try { // Apps Script의 302 리디렉션 응답을 처리하기 위해 mode: 'no-cors'를 사용합니다. await fetch(scriptUrl, { method: "POST", mode: "no-cors", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify(orderData) }); alert("✅ 주문이 정상 접수되었습니다!"); // 폼 초기화 document.getElementById("orderForm").reset(); } catch (error) { console.error("전송 에러:", error); alert("❌ 주문 접수 중 오류가 발생했습니다. 다시 시도해 주세요."); } }
